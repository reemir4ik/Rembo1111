export default { async fetch(request, env) { const ORIGIN = "https://reemir4ik.github.io"; // CORS preflight if (request.method === "OPTIONS") { return new Response(null, { headers: cors(ORIGIN) }); } const url = new URL(request.url); if (url.pathname !== "/generate") { return new Response("Not found", { status: 404, headers: cors(ORIGIN), }); } if (request.method !== "POST") { return new Response("Method not allowed", { status: 405, headers: cors(ORIGIN), }); } try { const body = await request.json(); const { brand, offer, utp, audience, goal, tone } = body; if (!offer) { return json({ error: "Offer required" }, 400); } const prompt = ` Ты копирайтер для рекламных баннеров (финансы/крипто). Тон: ${tone} Бренд: ${brand} Оффер: ${offer} УТП: ${utp} ЦА: ${audience} Цель: ${goal} Создай 3 разных варианта. Формат строго JSON: [ { "headline": "", "sub": "", "b1": "", "b2": "", "b3": "", "cta": "" } ] `.trim(); const ai = await fetch("https://api.openai.com/v1/responses", { method: "POST", headers: { "Authorization": `Bearer ${env.OPENAI_API_KEY}`, "Content-Type": "application/json", }, body: JSON.stringify({ model: "gpt-4.1-mini", input: prompt, max_output_tokens: 800 }), }); const data = await ai.json(); if (!ai.ok) { return json({ error: "OpenAI error", details: data }, 500); } let parsed; try { parsed = JSON.parse(data.output_text); } catch { return json({ error: "Invalid AI JSON", raw: data.output_text }, 500); } return json({ copies: parsed }, 200); } catch (err) { return json({ error: err.message }, 500); } function json(obj, status = 200) { return new Response(JSON.stringify(obj), { status, headers: { ...cors(ORIGIN), "Content-Type": "application/json" } }); } function cors(origin) { return { "Access-Control-Allow-Origin": origin, "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; } } };
